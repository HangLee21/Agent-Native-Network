cmake_minimum_required(VERSION 3.16)
project(ann_host_core C)
set(CMAKE_C_STANDARD 99)
add_library(ann_core src/ann.c src/ann_console.c src/ann_log.c)
target_include_directories(ann_core PUBLIC include PRIVATE src)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(ann_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(ann_host host/main.c)
target_include_directories(ann_host PRIVATE src)
target_link_libraries(ann_host PRIVATE ann_core)
add_executable(ann_c017_regressions host/c017_regressions.c)
target_link_libraries(ann_c017_regressions PRIVATE ann_core)
add_executable(ann_lifecycle_regressions host/lifecycle_regressions.c)
target_link_libraries(ann_lifecycle_regressions PRIVATE ann_core)
enable_testing()
add_test(NAME protocol_same_core COMMAND ann_host --scenario all)
