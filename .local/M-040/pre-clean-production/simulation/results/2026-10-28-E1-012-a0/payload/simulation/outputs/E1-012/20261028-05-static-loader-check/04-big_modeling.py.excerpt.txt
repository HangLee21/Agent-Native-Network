480:         device = list(device_map.values())[0]
481:         # `torch.Tensor.to(<int num>)` is not supported by `torch_npu` (see this [issue](https://github.com/Ascend/pytorch/issues/16)).
482:         if is_npu_available() and isinstance(device, int):
483:             device = f"npu:{device}"
484:         elif is_mlu_available() and isinstance(device, int):
485:             device = f"mlu:{device}"
486:         elif is_xpu_available() and isinstance(device, int):
487:             device = f"xpu:{device}"
488:         if device != "disk":
489:             model.to(device)
490:         else:
491:             raise ValueError(
492:                 "You are trying to offload the whole model to the disk. Please use the `disk_offload` function instead."
493:             )
494:     # Convert OrderedDict back to dict for easier usage
